# Irrigação Indoor com Dosagem Contínua em Litros

import numpy as np

# 1. Função de Ativação Contínua (Rampa Limitada entre 0 e Volume Máximo)
# Função 
def funcao_continua_litros(soma_ponderada, volume_maximo_litros=5.0):
    """
    Em vez de 0 ou 1, converte a soma ponderada em uma porcentagem contínua [0.0 a 1.0]
    e multiplica pelo volume máximo de irrigação permitido por ciclo.
    """
    # Restringe a ativação entre 0% (0.0) e 100% (1.0)
    porcentagem_vazao = max(0.0, min(soma_ponderada, 1.0))
    
    # Volume real dosado em Litros
    volume_litros = porcentagem_vazao * volume_maximo_litros
    
    return volume_litros, porcentagem_vazao

# 2. Normalização dos Dados de Entrada (Escala de 0.0 a 1.0 com unidades reais)
def normalizar_entradas(umid_solo, temp_ar, umid_ar, luz_ppfd, tempo_luz, evap):
    # Umidade do solo (0 a 100%): Invertida (solo mais seco = valor x1 maior)
    x1 = 1.0 - min(umid_solo / 100.0, 1.0)
    
    # Temperatura do ar (15 a 40 °C)
    x2 = min(max(temp_ar - 15, 0) / 25.0, 1.0)
    
    # Umidade relativa do ar (20 a 90%): Invertida (ar seco eleva transpiração)
    x3 = 1.0 - min(max(umid_ar - 20, 0) / 70.0, 1.0)
    
    # Intensidade luminosa PPFD (0 a 1200 µmol/m²/s)
    x4 = min(luz_ppfd / 1200.0, 1.0)
    
    # Tempo de luz ligada (0 a 18 Horas)
    x5 = min(tempo_luz / 18.0, 1.0)
    
    # Evapotranspiração da cultura (0 a 10 mm/dia)
    x6 = min(evap / 10.0, 1.0)
    
    return np.array([x1, x2, x3, x4, x5, x6])

# 3. Identificação da Fase do Cultivo
def identificar_fase_cultivo(tempo_luz):
    """Classifica se o ambiente favorece Vegetação ou Floração pelo fotoperíodo"""
    if tempo_luz >= 15:
        return "FASE VEGETATIVA", "Fotoperíodo longo (15h-18h) focado em crescimento de folhas e raízes."
    elif tempo_luz <= 13:
        return "FASE DE FLORAÇÃO", "Fotoperíodo curto (12h-13h) induzindo o florescimento."
    else:
        return "FASE DE TRANSIÇÃO", "Fotoperíodo intermediário."

# 4. Processamento Principal do Perceptron
def perceptron_volume_irrigacao(entradas, cap_max_litros=5.0):
    # Pesos atribuídos às 6 variáveis:
    # [Secura Solo, Temperatura, Ar Seco, PPFD, Tempo Luz, Evapotranspiração]
    pesos = np.array([0.35, 0.15, 0.15, 0.15, 0.10, 0.10])
    
    # Bias (Limiar de ativação):
    # Um bias negativo (-0.20) garante 0.0 Litros quando o solo já estiver bem úmido
    bias = -0.20
    
    # Soma Ponderada
    soma_ponderada = np.dot(entradas, pesos) + bias
    
    # Função de ativação contínua que gera o volume em Litros
    litros_dosados, pct_vazao = funcao_continua_litros(soma_ponderada, cap_max_litros)
    
    return litros_dosados, pct_vazao, soma_ponderada

# ==============================================================================
# CONTROLES INTERATIVOS DO GOOGLE COLAB
# ==============================================================================

# @markdown ### Ajuste as condições do ambiente e reservatório:
umidade_solo_pct = 60 # @param {type:"slider", min:0, max:100, step:1}
# @markdown *Umidade do Solo (em %)*

temperatura_ar_celsius = 33 # @param {type:"slider", min:15, max:40, step:1}
# @markdown *Temperatura do Ar (em °C)*

umidade_ar_pct = 40 # @param {type:"slider", min:20, max:90, step:1}
# @markdown *Umidade Relativa do Ar (em %)*

intensidade_luz_ppfd = 725 # @param {type:"slider", min:0, max:1200, step:25}
# @markdown *Intensidade Luminosa em PPFD (µmol/m²/s)*

tempo_luz_horas = 12 # @param {type:"slider", min:0, max:18, step:1}
# @markdown *Tempo de Luz Ligada (em Horas - Máx 18h)*

evapotranspiracao_mm_dia = 6 # @param {type:"slider", min:0, max:10, step:1}
# @markdown *Evapotranspiração estimada da cultura (em mm/dia)*

capacidade_maxima_litros = 3 # @param {type:"number"}
# @markdown *Volume Máximo de Rega por Ciclo (em Litros)*

# ==============================================================================
# EXECUÇÃO DO SISTEMA
# ==============================================================================

# Normalização e Cálculo
vetor_entradas = normalizar_entradas(
    umidade_solo_pct,
    temperatura_ar_celsius,
    umidade_ar_pct,
    intensidade_luz_ppfd,
    tempo_luz_horas,
    evapotranspiracao_mm_dia
)

litros_dosados, porcentagem_vazao, soma_interna = perceptron_volume_irrigacao(
    vetor_entradas, 
    capacidade_maxima_litros
)

fase_nome, fase_desc = identificar_fase_cultivo(tempo_luz_horas)

# Exibição do diagnóstico
print("=" * 68)
print("     SISTEMA DE IRRIGAÇÃO INDOOR - DOSAGEM CONTÍNUA EM LITROS     ")
print("=" * 68)

print(f"\nFASE DO CULTIVO IDENTIFICADA: {fase_nome}")
print(f"Detalhes: {fase_desc}")

print("-" * 68)
print(f"Soma Ponderada da Rede Neural: {soma_interna:.3f}")
print(f"Nível de Abertura da Válvula / Bomba: {porcentagem_vazao * 100:.1f}%")
print(f"\n>>> VOLUME DE IRRIGAÇÃO CALCULADO: {litros_dosados:.2f} LITROS <<<")

if litros_dosados == 0.0:
    print("Status: Nenhuma rega necessária. Solo suficientemente úmido.")
elif litros_dosados == capacidade_maxima_litros:
    print("Status: Alerta de solo seco / alto estresse! Solicitando dosagem MÁXIMA.")
else:
    print(f"Status: Aplicando rega proporcional de {litros_dosados:.2f} Litros.")

print("=" * 68)
